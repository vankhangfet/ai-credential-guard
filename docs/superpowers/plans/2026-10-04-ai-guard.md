> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Package npm `ai-guard` chặn credential (API key, password, private key, connection string...) trong prompt người dùng và file AI định đọc, cho 7 AI coding tool (Claude Code, Codex, OpenCode, Pi, Cline, Kiro, Copilot), kèm bypass có xác nhận + audit log.

**Architecture:** Engine TypeScript zero-dep duy nhất (regex catalog + entropy + sensitive-path matcher) chạy qua 2 lệnh chuẩn (`check-prompt`, `check-file`); 7 adapter mỏng đăng ký hook vào config native của từng tool; fallback Python mirror cho máy không npm. Spec: `docs/superpowers/specs/2026-10-04-ai-guard-credential-leak-prevention-design.md`.

**Tech Stack:** Node ≥18 (TypeScript, vitest, tsc — không runtime dependency), Python 3 stdlib cho fallback, GitHub Actions CI.

**Lưu ý scope:** Spec ước lượng ~80 rules; plan triển khai ~60 rules chất lượng cao trên 10 nhóm (ưu tiên precision; catalog là data nên mở rộng trivial).

```ts
const PLACEHOLDER_RE =
  /^(test|tests|example|examples|sample|samples|dummy|fake|placeholder|change[-_]?me|xxx+|\*+|your[-_][a-z0-9_.-]*|(?:your|my)[-_]?(?:key|token|pass(?:word)?|secret|api[_-]?key)[a-z0-9_-]*|<[^>]*>|\$\{[^}]*\}|\{\{[^}]*\}\}|\([a-z ]*\))[-_.a-z0-9]*$/i;

export function isPlaceholder(value: string): boolean {
  // guard chống ReDoS: placeholder là token ngắn; >200 ký tự coi như giá trị thật
  if (value.length > 200) return false;
  return PLACEHOLDER_RE.test(value.trim());
}

export function maskSecret(value: string): string {
  const s = value.trim();
  if (s.length <= 4) return "***";
  if (s.length <= 14) return s.slice(0, 2) + "***";
  return s.slice(0, 6) + "***" + s.slice(-4);
}
```

```ts
export function shannonEntropy(s: string): number {
  if (!s.length) return 0;
  const freq: Record<string, number> = {};
  for (const ch of s) freq[ch] = (freq[ch] ?? 0) + 1;
  let h = 0;
  for (const c of Object.values(freq)) {
    const p = c / s.length;
    h -= p * Math.log2(p);
  }
  return h;
}

const COMMON_WORD_RE = /^[a-z][a-z\-']{5,}$/;

export function isHighEntropyToken(s: string, minLen = 20, minH = 3.5): boolean {
  if (s.length < minLen) return false;
  // bỏ token thuần chữ thường giống từ tự nhiên
  if (COMMON_WORD_RE.test(s) && shannonEntropy(s) < 4.2) return false;
  return shannonEntropy(s) >= minH;
}
```

```ts
import { join } from "node:path";
```

```ts
import { builtInRules, compileRules } from "./rules";

function readJsonSafe(path: string): unknown | null {
  try {
    if (!existsSync(path)) return null;
    return JSON.parse(readFileSync(path, "utf8"));
  } catch {
    return null;
  }
}

export function loadRules(projectRoot: string): CompiledRule[] {
  const raw = readJsonSafe(join(projectRoot, ".ai-guard", "rules.json")) as ProjectRules | null;
  let rules: Rule[] = [...builtInRules];
  if (raw) {
    if (Array.isArray(raw.add)) rules = rules.concat(raw.add.filter((r) => r && r.id && r.pattern !== undefined));
    if (raw.override) {
      rules = rules.map((r) => (raw.override![r.id]?.severity ? { ...r, severity: raw.override![r.id].severity! } : r));
    }
    if (Array.isArray(raw.remove)) {
      const gone = new Set(raw.remove);
      rules = rules.filter((r) => !gone.has(r.id));
    }
  }
  return compileRules(rules).filter((r) => r.pattern === "" ? true : !!r.re);
}

export function loadConfig(projectRoot: string): AiGuardConfig {
  const cfg = (readJsonSafe(join(projectRoot, ".ai-guard", "config.json")) ?? {}) as AiGuardConfig;
  return {
    sensitivePaths: Array.isArray(cfg.sensitivePaths) && cfg.sensitivePaths.length ? cfg.sensitivePaths : DEFAULT_SENSITIVE_PATHS,
    genericSecret: cfg.genericSecret ?? true,
    highEntropy: cfg.highEntropy ?? true,
  };
}
```

```ts
import type { AiGuardConfig, CompiledRule, Finding } from "../types";
import { isPlaceholder, maskSecret } from "./mask";
import { isHighEntropyToken } from "./entropy";

const GENERIC_ASSIGN_RE =
  /\b(passwo?rd|passwd|secret|api[_-]?key|apikey|auth[_-]?token|access[_-]?token|client[_-]?secret|admin[_-]?pass)\b\s*[:=]\s*["']?([^\s"']{8,})["']?/gi;
const TOKEN_CANDIDATE_RE = /[A-Za-z0-9+/_$&!#%-]{20,}/g;
const MAX_FINDINGS = 20;

export function scanText(text: string, rules: CompiledRule[], cfg: AiGuardConfig): Finding[] {
  const findings: Finding[] = [];
  for (const rule of rules) {
    if (!rule.re) continue;
    rule.re.lastIndex = 0;
    const m = rule.re.exec(text);
    if (m && m[0]) {
      findings.push({ ruleId: rule.id, severity: rule.severity, description: rule.description, preview: maskSecret(m[0]) });
      if (findings.length >= MAX_FINDINGS) return findings;
    }
  }
  if (cfg.genericSecret) {
    GENERIC_ASSIGN_RE.lastIndex = 0;
    let m: RegExpExecArray | null;
    while ((m = GENERIC_ASSIGN_RE.exec(text)) !== null) {
      const value = m[2];
      if (!isPlaceholder(value) && isHighEntropyToken(value, 8, 2.5)) {
        findings.push({ ruleId: "generic-secret", severity: "warn", description: "Gán giá trị secret trong text", preview: maskSecret(value) });
        if (findings.length >= MAX_FINDINGS) return findings;
      }
    }
  }
  if (cfg.highEntropy) {
    const seen = new Set<string>();
    for (const token of text.match(TOKEN_CANDIDATE_RE) ?? []) {
      if (seen.has(token)) continue;
      seen.add(token);
      if (isPlaceholder(token)) continue;
      if (isHighEntropyToken(token)) {
        findings.push({ ruleId: "high-entropy", severity: "warn", description: "Token entropy cao", preview: maskSecret(token) });
        if (findings.length >= MAX_FINDINGS) return findings;
      }
    }
  }
  return findings;
}
```

```ts
export function normalizePath(p: string): string {
  return p.replace(/\\/g, "/").replace(/^\.\//, "").replace(/^\/+/, "").toLowerCase();
}

export function globToRegex(glob: string): RegExp {
  const n = normalizePath(glob);
  let out = "";
  for (let i = 0; i < n.length; i++) {
    const c = n[i];
    if (c === "*") {
      if (n[i + 1] === "*") { out += ".*"; i++; }
      else out += "[^/]*";
    } else if (c === "?") out += "[^/]";
    else out += c.replace(/[.+^${}()|[\]\\]/g, "\\$&");
  }
  return new RegExp("^" + out + "$", "i");
}

export function isSensitivePath(path: string, patterns: string[]): boolean {
  const rel = normalizePath(path);
  // so khớp cả từng hậu tố thư mục (config/.env khớp pattern .env)
  for (const pat of patterns) {
    const re = globToRegex(pat);
    if (re.test(rel)) return true;
    // cho pattern không có '/': khớp cả khi file nằm trong subdir
    if (!pat.includes("/")) {
      const base = rel.split("/").pop()!;
      if (re.test(base)) return true;
    }
  }
  return false;
}
```

```ts
import { createHash } from "node:crypto";
import { existsSync, readdirSync, readFileSync, rmSync, writeFileSync } from "node:fs";
import { join } from "node:path";

interface BypassFlag {
  scope: "prompt" | "file";
  pathHash?: string;
  grantedAt: string;
  expiresAt: number; // epoch ms
  duration_min: number;
}

const bypassDir = (root: string) => join(root, ".ai-guard", ".bypass");

function flagName(scope: string, path?: string): string {
  const h = path ? createHash("sha1").update(path.replace(/\\/g, "/").toLowerCase()).digest("hex").slice(0, 12) : "all";
  return `${scope}-${h}.json`;
}

export function grantBypass(root: string, scope: "prompt" | "file", path: string | undefined, minutes: number): { duration_min: number } {
  const dir = bypassDir(root);
  existsSync(dir) || (() => { throw new Error("no .ai-guard dir"); })(); // caller đảm bảo root hợp lệ; nếu lỗi -> throw để command bắt
  const now = Date.now();
  const flag: BypassFlag = {
    scope,
    pathHash: path ? flagName(scope, path).split("-")[1] : undefined,
    grantedAt: new Date(now).toISOString(),
    expiresAt: now + minutes * 60_000,
    duration_min: minutes,
  };
  writeFileSync(join(dir, flagName(scope, path)), JSON.stringify(flag));
  return { duration_min: minutes };
}

export function hasValidBypass(root: string, scope: "prompt" | "file", path?: string): boolean {
  const dir = bypassDir(root);
  try {
    if (!existsSync(dir)) return false;
    // dọn flag hết hạn
    for (const f of readdirSync(dir)) {
      if (!f.endsWith(".json")) continue;
      try {
        const flag = JSON.parse(readFileSync(join(dir, f), "utf8")) as BypassFlag;
        if (flag.expiresAt <= Date.now()) rmSync(join(dir, f));
      } catch { rmSync(join(dir, f)); }
    }
    const target = join(dir, flagName(scope, path));
    if (!existsSync(target)) return false;
    const flag = JSON.parse(readFileSync(target, "utf8")) as BypassFlag;
    return flag.scope === scope && flag.expiresAt > Date.now();
  } catch {
    return false;
  }
}
```

```ts
function extractPaths(argv: string[], input: string): { abs: string; rel: string } | null {
  // 1) argv: --root X ... <path> (path là đối số cuối không bắt đầu bằng --)
  const positional = argv.filter((a, i) => !a.startsWith("--") && !(argv[i - 1] === "--root" || argv[i - 1] === "--tool") && i > 0);
  // 2) stdin JSON: tool_input.file_path / path / notebook_path
  let fromStdin: string | null = null;
  const s = input.trim();
  if (s.startsWith("{")) {
    try {
      const j = JSON.parse(s);
      const p = j?.tool_input?.file_path ?? j?.tool_input?.notebook_path ?? j?.tool_input?.path ?? j?.path;
      if (typeof p === "string" && p) fromStdin = p;
    } catch { /* ignore */ }
  }
  const raw = positional[positional.length - 1] ?? fromStdin;
  if (!raw) return null;
  return { abs: raw, rel: raw };
}

export async function checkFile(argv: string[], input: string): Promise<number> {
  const rootIdx = argv.indexOf("--root");
  const cwdArg = rootIdx >= 0 ? argv[rootIdx + 1] : process.cwd();
  const toolIdx = argv.indexOf("--tool");
  const tool = toolIdx >= 0 ? argv[toolIdx + 1] : "unknown";
  const { findProjectRoot } = await import("../root");
  const projectRoot = findProjectRoot(cwdArg);
  if (!projectRoot) return 0;
  const p = extractPaths(argv, input);
  if (!p) return 0;
  const abs = isAbsolute(p.abs) ? p.abs : join(projectRoot, p.abs);
  const rel = normalizePath(relative(projectRoot, abs) || p.rel);
  const cfg = loadConfig(projectRoot);
  if (!isSensitivePath(rel, cfg.sensitivePaths ?? [])) return 0;
  if (hasValidBypass(projectRoot, "file", rel)) {
    appendAuditEvent(projectRoot, { ts: new Date().toISOString(), tool, event: "file", action: "allowed_after_confirm", path: rel });
    return 0;
  }
  appendAuditEvent(projectRoot, { ts: new Date().toISOString(), tool, event: "file", action: "blocked", path: rel });
  process.stderr.write(

      "ai-guard: ĐÃ CHẶN — file nhạy cảm (chưa cho AI đọc/ghi).",
```

```ts
import { existsSync, readFileSync, writeFileSync } from "node:fs";
import { join } from "node:path";
import type { AdapterBase, InstallOptions, InstallResult } from "./types";
import { MARKER, readJson, writeJson } from "../util/json-config";

interface HookEntry { matcher?: string; hooks: Array<{ type: string; command: string }> }
interface ClaudeSettings { permissions?: unknown; hooks?: Record<string, HookEntry[]>; [k: string]: unknown }

const CMD_PROMPT = "npx --no-install ai-guard check-prompt --tool claude-code";
const CMD_FILE = "npx --no-install ai-guard check-file --tool claude-code";

function entriesWithMarker(list: HookEntry[] | undefined, cmdPart: string): boolean {
  return !!list?.some((e) => e.hooks.some((h) => h.command.includes(MARKER) && h.command.includes(cmdPart)));
}

function appendInstruction(root: string): void {
  const file = join(root, "CLAUDE.md");
  const template = readFileSync(require.resolve("../../templates/claude-instructions.md"), "utf8");
  const current = existsSync(file) ? readFileSync(file, "utf8") : "";
  if (current.includes("<!-- ai-guard:start -->")) return;
  writeFileSync(file, current + (current && !current.endsWith("\n") ? "\n" : "") + template);
}

export const claudeCodeAdapter: AdapterBase = {
  id: "claude-code",
  label: "Claude Code",
  detect: (root) => existsSync(join(root, ".claude")),
  install(root: string, opts: InstallOptions): InstallResult {
    try {
      const file = join(root, ".claude", "settings.json");
      const settings = readJson<ClaudeSettings>(file, {});
      settings.hooks ??= {};
      settings.hooks.UserPromptSubmit ??= [];
      settings.hooks.PreToolUse ??= [];
      if (!entriesWithMarker(settings.hooks.UserPromptSubmit, "check-prompt")) {
        settings.hooks.UserPromptSubmit.push({ hooks: [{ type: "command", command: CMD_PROMPT }] });
      }
      if (!entriesWithMarker(settings.hooks.PreToolUse, "check-file")) {
        settings.hooks.PreToolUse.push({ matcher: "Read|Glob|Grep", hooks: [{ type: "command", command: CMD_FILE }] });
      }
      writeJson(file, settings);
      if (opts.instructions) appendInstruction(root);
      return { adapter: "claude-code", ok: true, detail: "hooks UserPromptSubmit + PreToolUse đã đăng ký" };
    } catch (e) {
      return { adapter: "claude-code", ok: false, detail: String(e) };
    }
  },
  uninstall(root: string): InstallResult {
    try {
      const file = join(root, ".claude", "settings.json");
      const settings = readJson<ClaudeSettings>(file, {});
      if (settings.hooks) {
        for (const ev of Object.keys(settings.hooks)) {
          settings.hooks[ev] = (settings.hooks[ev] ?? []).filter((e) => !e.hooks.some((h) => h.command.includes(MARKER)));
          if (settings.hooks[ev].length === 0) delete settings.hooks[ev];
        }
        if (Object.keys(settings.hooks).length === 0) delete settings.hooks;
      }
      writeJson(file, settings);
      return { adapter: "claude-code", ok: true, detail: "hooks ai-guard đã gỡ" };
    } catch (e) {
      return { adapter: "claude-code", ok: false, detail: String(e) };
    }
  },
  doctor(root: string) {
    const settings = readJson<ClaudeSettings>(join(root, ".claude", "settings.json"), {});
    const ok = entriesWithMarker(settings.hooks?.UserPromptSubmit, "check-prompt")
      && entriesWithMarker(settings.hooks?.PreToolUse, "check-file");
    return { ok, detail: ok ? "hooks đã đăng ký" : "thiếu hook ai-guard trong .claude/settings.json" };
  },
};
```

```ts
import { appendFileSync, existsSync, mkdirSync, writeFileSync } from "node:fs";
import { join } from "node:path";
import { allAdapters } from "../adapters";
import type { InstallResult } from "../adapters/types";

const DEFAULT_RULES = {
  add: [{ id: "example-internal", severity: "block", pattern: "CORP-[A-Z0-9]{8,}", description: "Ví dụ: đổi thành format token nội bộ của bạn" }],
  override: {},
  remove: [],
};
const DEFAULT_CONFIG = { sensitivePaths: null, genericSecret: true, highEntropy: true };

export async function init(argv: string[], _input: string): Promise<number> {
  const rootIdx = argv.indexOf("--root");
  const root = rootIdx >= 0 ? argv[rootIdx + 1] : process.cwd();
  const noInstructions = argv.includes("--no-instructions");
  const toolsIdx = argv.indexOf("--tools");
  const wanted = toolsIdx >= 0 ? argv[toolsIdx + 1].split(",") : null;

  const adapters = allAdapters();
  const detected = adapters.filter((a) => wanted ? wanted.includes(a.id) : a.detect(root));
  if (!detected.length) {
    process.stderr.write([
      "ai-guard: không phát hiện AI tool nào trong dự án.",
      "Dấu hiệu được kiểm tra: .claude, .codex, .opencode, .pi, .cline, .kiro, .vscode",
      "Dùng: npx ai-guard init --tools claude-code,codex,opencode,pi,cline,kiro,copilot",
    ].join("\n") + "\n");
    return 1;
  }

  // 1) cấu trúc .ai-guard
  mkdirSync(join(root, ".ai-guard", "logs"), { recursive: true });
  mkdirSync(join(root, ".ai-guard", ".bypass"), { recursive: true });
  const rulesFile = join(root, ".ai-guard", "rules.json");
  const configFile = join(root, ".ai-guard", "config.json");
  if (!existsSync(rulesFile)) writeFileSync(rulesFile, JSON.stringify(DEFAULT_RULES, null, 2) + "\n");
  if (!existsSync(configFile)) writeFileSync(configFile, JSON.stringify(DEFAULT_CONFIG, null, 2) + "\n");

  // 2) .gitignore
  const giPath = join(root, ".gitignore");
  const gi = existsSync(giPath) ? readFileSyncText(giPath) : "";
  const additions = [".ai-guard/logs/", ".ai-guard/.bypass/"].filter((l) => !gi.includes(l));
  if (additions.length) appendFileSync(giPath, (gi && !gi.endsWith("\n") ? "\n" : "") + "# ai-guard\n" + additions.join("\n") + "\n");

  // 3) install adapters
  const results: InstallResult[] = [];
  for (const a of detected) {
    results.push(a.install(root, { instructions: !noInstructions }));
  }

  // 4) summary
  console.log("ai-guard: đã khởi tạo.");
  for (const r of results) console.log(`  ${r.ok ? "✓" : "✗"} ${r.adapter}: ${r.detail}`);
  console.log("  Cấu hình rule: .ai-guard/rules.json — audit log: .ai-guard/logs/");
  console.log("  Kiểm tra: npx ai-guard doctor");
  console.log("  Lưu ý: thêm ai-guard vào devDependencies (npm i -D ai-guard) để hook chạy mà không cần mạng.");
  return results.some((r) => !r.ok) ? 1 : 0;
}

function readFileSyncText(p: string): string {
  return require("node:fs").readFileSync(p, "utf8");
}
```

```ts
export function selfTest(): { total: number; failures: string[] } {
  const rules = compileRules(builtInRules);
  const cfg = { genericSecret: true, highEntropy: true, sensitivePaths: DEFAULT_SENSITIVE_PATHS };
  const failures: string[] = [];
  for (const f of FIXTURES) {
    const findings = scanText(f.input, rules, cfg);
    const blocked = findings.some((x) => x.severity === "block");
    const ruleOk = f.expectRule ? findings.some((x) => x.ruleId === f.expectRule) : true;
    if (blocked !== f.expectBlocked || !ruleOk) {
```

```ts
import { existsSync, readFileSync, writeFileSync } from "node:fs";
import { join } from "node:path";
import type { AdapterBase } from "./types";
import { readJson, writeJson, MARKER } from "../util/json-config";

const CMD_PROMPT = "npx --no-install ai-guard check-prompt --tool codex";
const CMD_FILE = "npx --no-install ai-guard check-file --tool codex";

interface CodexHook { type: string; command: string }
interface CodexEvent { matcher?: string; hooks: CodexHook[] }
interface CodexHooksFile { hooks: Record<string, CodexEvent[]> }

const INSTRUCTIONS = `<!-- ai-guard:start -->
## Bảo mật credential (ai-guard)
Không yêu cầu/đọc credential thật (.env, *.pem, id_rsa, credentials*). Dùng .env.example.
<!-- ai-guard:end -->
`;

function withMarker(list: CodexEvent[] | undefined, cmdPart: string) {
  return !!list?.some((e) => e.hooks.some((h) => h.command.includes(MARKER) && h.command.includes(cmdPart)));
}

export const codexAdapter: AdapterBase = {
  id: "codex",
  label: "Codex CLI",
  detect: (root) => existsSync(join(root, ".codex")),
  install(root) {
    try {
      const file = join(root, ".codex", "hooks.json");
      const data = readJson<CodexHooksFile>(file, { hooks: {} });
      data.hooks ??= {};
      data.hooks.UserPromptSubmit ??= [];
      data.hooks.PreToolUse ??= [];
      if (!withMarker(data.hooks.UserPromptSubmit, "check-prompt"))
        data.hooks.UserPromptSubmit.push({ hooks: [{ type: "command", command: CMD_PROMPT }] });
      if (!withMarker(data.hooks.PreToolUse, "check-file"))
        data.hooks.PreToolUse.push({ matcher: "read|grep|glob", hooks: [{ type: "command", command: CMD_FILE }] });
      writeJson(file, data);
      // education
      const agentsMd = join(root, "AGENTS.md");
      const cur = existsSync(agentsMd) ? readFileSync(agentsMd, "utf8") : "";
      if (!cur.includes("ai-guard:start")) writeFileSync(agentsMd, cur + INSTRUCTIONS);
      return { adapter: "codex", ok: true, detail: "hooks.json đã đăng ký" };
    } catch (e) { return { adapter: "codex", ok: false, detail: String(e) }; }
  },
  uninstall(root) {
    try {
      const file = join(root, ".codex", "hooks.json");
      const data = readJson<CodexHooksFile>(file, { hooks: {} });
      if (data.hooks) {
        for (const ev of Object.keys(data.hooks)) {
          data.hooks[ev] = (data.hooks[ev] ?? []).filter((e) => !e.hooks.some((h) => h.command.includes(MARKER)));
          if (!data.hooks[ev].length) delete data.hooks[ev];
        }
      }
      writeJson(file, data);
      return { adapter: "codex", ok: true, detail: "đã gỡ" };
    } catch (e) { return { adapter: "codex", ok: false, detail: String(e) }; }
  },
  doctor(root) {
    const data = readJson<CodexHooksFile>(join(root, ".codex", "hooks.json"), { hooks: {} });
    const ok = withMarker(data.hooks?.UserPromptSubmit, "check-prompt") && withMarker(data.hooks?.PreToolUse, "check-file");
    return { ok, detail: ok ? "hooks đã đăng ký" : "thiếu hook trong .codex/hooks.json" };
  },
};
```

```ts
  const r = spawnSync("npx", ["--no-install", "ai-guard", ...args], {
```

```python
#!/usr/bin/env python3
"""ai-guard fallback engine (Python mirror). Cùng contract với CLI Node:
check-prompt / check-file / allow. Exit: 0 cho qua, 2 chặn."""
import argparse
import datetime
import glob as globmod
import hashlib
import json
import math
import os
import re
import sys

DEFAULT_SENSITIVE_PATHS = [
    ".env", ".env.*", "*.env", "*.pem", "*.key", "*.p12", "*.pfx",
    "id_rsa*", "id_ed25519*", "id_ecdsa*", "id_dsa*",
    "credentials*", "*credentials*.json", "*credentials*.yaml", "*credentials*.yml",
    "secrets/**", ".secrets/**", ".aws/credentials", ".npmrc", ".netrc", ".pypirc",
    "service-account*.json",
]
PLACEHOLDER_RE = re.compile(
    r"^(test|tests|example|examples|sample|samples|dummy|fake|placeholder|changeme|change[-_]?me|x+|\*+|your[-_][a-z0-9_.-]*|(?:your|my)[-_]?(?:key|token|pass(?:word)?|secret|api[_-]?key)[a-z0-9_-]*|<[^>]*>|\$\{[^}]*\})[-_.a-z0-9]*$",
    re.I)
GENERIC_ASSIGN_RE = re.compile(
    r"\b(passwo?rd|passwd|secret|api[_-]?key|apikey|auth[_-]?token|access[_-]?token|client[_-]?secret|admin[_-]?pass)\b\s*[:=]\s*[\"']?([^\s\"']{8,})[\"']?",
    re.I)

def mask_secret(s: str) -> str:
    s = s.strip()
    if len(s) <= 4:
        return "***"
    if len(s) <= 14:
        return s[:2] + "***"
    return s[:6] + "***" + s[-4:]

def is_placeholder(v: str) -> bool:
    if len(v) > 200:  # guard chống backtracking, đồng bộ với bản TS
        return False
    return bool(PLACEHOLDER_RE.match(v.strip()))

def shannon_entropy(s: str) -> float:
    if not s:
        return 0.0
    freq = {}
    for ch in s:
        freq[ch] = freq.get(ch, 0) + 1
    n = len(s)
    return -sum((c / n) * math.log2(c / n) for c in freq.values())

def is_high_entropy(s: str, min_len=20, min_h=3.5) -> bool:
    if len(s) < min_len:
        return False
    return shannon_entropy(s) >= min_h

def builtin_rules_path():
    p = os.environ.get("AI_GUARD_BUILTIN_RULES")
    if p and os.path.exists(p):
        return p
    return os.path.join(os.path.dirname(os.path.abspath(__file__)), "rules.json")

def load_rules(root: str):
    rules = []
    try:
        with open(builtin_rules_path(), encoding="utf8") as f:
            rules = json.load(f)
    except Exception:
        rules = []
    try:
        with open(os.path.join(root, ".ai-guard", "rules.json"), encoding="utf8") as f:
            proj = json.load(f)
        rules += proj.get("add", [])
        for rid, ov in (proj.get("override") or {}).items():
            for r in rules:
                if r["id"] == rid and "severity" in ov:
                    r["severity"] = ov["severity"]
        gone = set(proj.get("remove", []))
        rules = [r for r in rules if r["id"] not in gone]
    except Exception:
        pass
    out = []
    for r in rules:
        if not r.get("pattern"):
            out.append(r)
            continue
        try:
            out.append({**r, "re": re.compile(r["pattern"])})
        except Exception:
            continue
    return out

def glob_to_regex(g: str):
    g = g.replace("\\", "/").lstrip("./").lower()
    out, i = "", 0
    while i < len(g):
        c = g[i]
        if c == "*":
            if i + 1 < len(g) and g[i + 1] == "*":
                out += ".*"; i += 1
            else:
                out += "[^/]*"
        elif c == "?":
            out += "[^/]"
        else:
            out += re.escape(c)
        i += 1
    return re.compile("^" + out + "$", re.I)

def is_sensitive_path(rel: str, patterns) -> bool:
    rel = rel.replace("\\", "/").lstrip("./").lower()
    for pat in patterns:
        rx = glob_to_regex(pat)
        if rx.match(rel):
            return True
        if "/" not in pat and rx.match(rel.split("/")[-1]):
            return True
    return False

def scan(text: str, rules, generic=True, entropy=True):
    findings = []
    for r in rules:
        rx = r.get("re")
        if rx is None:
            continue
        m = rx.search(text)
        if m:
            findings.append({"ruleId": r["id"], "severity": r["severity"], "preview": mask_secret(m.group(0))})
    if generic:
        for m in GENERIC_ASSIGN_RE.finditer(text):
            val = m.group(2)
            if not is_placeholder(val) and shannon_entropy(val) >= 2.5:
                findings.append({"ruleId": "generic-secret", "severity": "warn", "preview": mask_secret(val)})
    if entropy:
        seen = set()
        for tok in re.findall(r"[A-Za-z0-9+/_$&!#%-]{20,}", text):
            if tok in seen or is_placeholder(tok):
                continue
            seen.add(tok)
            if is_high_entropy(tok):
                findings.append({"ruleId": "high-entropy", "severity": "warn", "preview": mask_secret(tok)})
    return findings[:20]

def append_audit(root, evt):
    try:
        d = os.path.join(root, ".ai-guard", "logs")
        os.makedirs(d, exist_ok=True)
        day = datetime.datetime.now().strftime("%Y-%m-%d")
        with open(os.path.join(d, day + ".jsonl"), "a", encoding="utf8") as f:
            f.write(json.dumps(evt, ensure_ascii=False) + "\n")
    except Exception:
        pass

def find_root(start: str):
    env = os.environ.get("AI_GUARD_ROOT")
    if env and os.path.isdir(env):
        return env
    d = os.path.abspath(start)
    for _ in range(30):
        if os.path.isdir(os.path.join(d, ".ai-guard")):
            return d
        nd = os.path.dirname(d)
        if nd == d:
            return None
        d = nd
    return None

def bypass_ok(root, scope, path=None):
    d = os.path.join(root, ".ai-guard", ".bypass")
    key = hashlib.sha1((path or "").replace("\\", "/").lower().encode()).hexdigest()[:12] if path else "all"
    target = os.path.join(d, f"{scope}-{key}.json")
    try:
        for f in globmod.glob(os.path.join(d, "*.json")):
            try:
                flag = json.load(open(f, encoding="utf8"))
                if flag.get("expiresAt", 0) <= datetime.datetime.now().timestamp() * 1000:
                    os.remove(f)
            except Exception:
                os.remove(f)
        if not os.path.exists(target):
            return False
        flag = json.load(open(target, encoding="utf8"))
        return flag.get("scope") == scope and flag.get("expiresAt", 0) > datetime.datetime.now().timestamp() * 1000
    except Exception:
        return False

def now_iso():
    return datetime.datetime.now().astimezone().isoformat(timespec="seconds")

def extract_prompt(stdin_text):
    s = stdin_text.strip()
    if not s:
        return None
    if s.startswith("{"):
        try:
            j = json.loads(s)
            p = j.get("prompt") or j.get("input") or j.get("message") or j.get("text")
            return p if isinstance(p, str) else None
        except Exception:
            pass
    return s

def cmd_check_prompt(args, stdin_text):
    root = find_root(args.root)
    prompt = extract_prompt(stdin_text)
    if not root or not prompt:
        return 0
    rules = load_rules(root)
    findings = scan(prompt, rules)
    warns = [f for f in findings if f["severity"] == "warn"]
    blocks = [f for f in findings if f["severity"] == "block"]
    if warns:
        append_audit(root, {"ts": now_iso(), "tool": args.tool, "event": "prompt", "action": "warned",
                            "rule": warns[0]["ruleId"], "preview": warns[0]["preview"], "count": len(warns)})
    if not blocks:
        return 0
    if bypass_ok(root, "prompt"):
        append_audit(root, {"ts": now_iso(), "tool": args.tool, "event": "prompt", "action": "allowed_after_confirm",
                            "rule": blocks[0]["ruleId"], "preview": blocks[0]["preview"], "count": len(blocks)})
        return 0
    append_audit(root, {"ts": now_iso(), "tool": args.tool, "event": "prompt", "action": "blocked",
                        "rule": blocks[0]["ruleId"], "preview": blocks[0]["preview"], "count": len(blocks)})
    sys.stderr.write("ai-guard: ĐÃ CHẶN — phát hiện credential trong prompt.\n")
    for f in blocks[:5]:
        sys.stderr.write(f"  • {f['ruleId']} [preview: {f['preview']}]\n")
    sys.stderr.write("Nếu CỐ Ý gửi: chạy lệnh tương đương `python engine.py allow prompt --5m` rồi gửi lại.\n")
    return 2

def cmd_check_file(args, stdin_text):
    root = find_root(args.root)
    if not root:
        return 0
    path = args.path
    if not path and stdin_text.strip().startswith("{"):
        try:
            j = json.loads(stdin_text)
            ti = j.get("tool_input") or {}
            path = ti.get("file_path") or ti.get("notebook_path") or ti.get("path") or j.get("path")
        except Exception:
            path = None
    if not path:
        return 0
    abs_path = path if os.path.isabs(path) else os.path.join(root, path)
    rel = os.path.relpath(abs_path, root).replace("\\", "/")
    if not is_sensitive_path(rel, DEFAULT_SENSITIVE_PATHS):
        return 0
    if bypass_ok(root, "file", rel.lower()):
        append_audit(root, {"ts": now_iso(), "tool": args.tool, "event": "file", "action": "allowed_after_confirm", "path": rel})
        return 0
    append_audit(root, {"ts": now_iso(), "tool": args.tool, "event": "file", "action": "blocked", "path": rel})
    sys.stderr.write(f"ai-guard: ĐÃ CHẶN — file nhạy cảm: {rel}\n")
    sys.stderr.write(f"Cho phép file này: `python engine.py allow file {rel} --10m`\n")
    return 2

def cmd_allow(args):
    root = find_root(args.root)
    if not root:
        sys.stderr.write("ai-guard: không tìm thấy .ai-guard — chạy installer trước.\n")
        return 1
    if args.scope not in ("prompt", "file"):
        sys.stderr.write("ai-guard: scope phải là prompt hoặc file\n")
        return 1
    minutes = args.duration
    if not minutes:
        sys.stderr.write("ai-guard: thiếu --Nm (vd --5m)\n")
        return 1
    mins = int(minutes[2:-1] if minutes.endswith("m") else minutes[2:])
    d = os.path.join(root, ".ai-guard", ".bypass")
    os.makedirs(d, exist_ok=True)
    key = hashlib.sha1((args.path or "").replace("\\", "/").lower().encode()).hexdigest()[:12] if args.path else "all"
    expires = (datetime.datetime.now() + datetime.timedelta(minutes=mins)).timestamp() * 1000
    with open(os.path.join(d, f"{args.scope}-{key}.json"), "w", encoding="utf8") as f:
        json.dump({"scope": args.scope, "pathHash": key if args.path else None,
                   "grantedAt": now_iso(), "expiresAt": expires, "duration_min": mins}, f)
    append_audit(root, {"ts": now_iso(), "tool": "user", "event": "bypass", "action": "bypass_granted",
                        "scope": args.scope, "path": args.path, "duration_min": mins})
    print(f"ai-guard: cho phép {args.scope} {args.path or ''} trong {mins} phút (audit đã ghi).")
    return 0

def main():
    p = argparse.ArgumentParser(prog="ai-guard")
    sub = p.add_subparsers(dest="cmd", required=True)
    cp = sub.add_parser("check-prompt"); cp.add_argument("--root", default="."); cp.add_argument("--tool", default="unknown")
    cf = sub.add_parser("check-file"); cf.add_argument("--root", default="."); cf.add_argument("--tool", default="unknown"); cf.add_argument("path", nargs="?")
    al = sub.add_parser("allow"); al.add_argument("scope"); al.add_argument("path", nargs="?"); al.add_argument("--root", default="."); al.add_argument("duration", nargs="?")
    args = p.parse_args()
    stdin_text = sys.stdin.read() if not sys.stdin.isatty() else ""
    if args.cmd == "check-prompt":
        sys.exit(cmd_check_prompt(args, stdin_text))
    elif args.cmd == "check-file":
        sys.exit(cmd_check_file(args, stdin_text))
    elif args.cmd == "allow":
        sys.exit(cmd_allow(args))

if __name__ == "__main__":
    main()
```
